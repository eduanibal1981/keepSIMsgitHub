// send-admin-email — sends one hand-written admin message through Resend.
//
// Composition happens in keep_sims_admin; `admin_send_email()` validates and
// queues a row; this delivers it. The split matters: the RPC runs inside the
// caller's transaction and must not depend on a third-party HTTP call
// succeeding, and the Resend key must never be reachable from a client.
//
// TWO THINGS THIS FUNCTION WILL NOT DO, both deliberate:
//
//  1. It does not accept HTML from the admin. The body is plain text, escaped
//     here, and wrapped in a fixed template. An admin form that renders raw
//     HTML from a verified domain is a phishing kit the moment a power-user
//     session leaks — and escaping is also what stops a perfectly innocent
//     "Q&A <10GB" from breaking the message.
//
//  2. It does not take the sender from the request. `from` is read from the
//     queued row, which admin_send_email() already checked against the
//     ADMIN_EMAIL_FROM_OPTIONS allow-list.

import { serve } from "https://deno.land/std@0.168.0/http/server.ts";
import { createClient } from "https://esm.sh/@supabase/supabase-js@2";

const corsHeaders = {
  "Access-Control-Allow-Origin": "*",
  "Access-Control-Allow-Headers":
    "authorization, x-client-info, apikey, content-type, x-sync-secret",
};

function escapeHtml(s: string): string {
  return s
    .replace(/&/g, "&amp;")
    .replace(/</g, "&lt;")
    .replace(/>/g, "&gt;")
    .replace(/"/g, "&quot;")
    .replace(/'/g, "&#39;");
}

/** Plain text -> a readable, deliberately plain HTML body. */
function renderHtml(bodyText: string, fromName: string): string {
  // Blank-line separated paragraphs, single newlines as line breaks. Nothing
  // clever: a business email that looks like a marketing template reads as one.
  const paragraphs = escapeHtml(bodyText)
    .split(/\n\s*\n/)
    .map((p) => p.trim())
    .filter((p) => p.length > 0)
    .map((p) => `<p style="margin:0 0 16px 0;">${p.replace(/\n/g, "<br>")}</p>`)
    .join("");

  return `<div style="font-family:-apple-system,Segoe UI,Roboto,Helvetica,Arial,sans-serif;font-size:15px;line-height:1.6;color:#1f2328;max-width:640px;">
${paragraphs}
<div style="margin-top:28px;padding-top:14px;border-top:1px solid #e4e7eb;font-size:12px;color:#6b7280;">${escapeHtml(fromName)}</div>
</div>`;
}

serve(async (req) => {
  if (req.method === "OPTIONS") {
    return new Response("ok", { headers: corsHeaders });
  }

  const supabase = createClient(
    Deno.env.get("SUPABASE_URL") ?? "",
    Deno.env.get("SUPABASE_SERVICE_ROLE_KEY") ?? "",
  );

  let emailId: string | null = null;

  try {
    const { data: expected } = await supabase
      .rpc("get_internal_secret", { p_name: "admin_email_secret" });
    if (!expected || (req.headers.get("x-sync-secret") ?? "") !== expected) {
      return new Response(JSON.stringify({ error: "forbidden" }), {
        status: 403,
        headers: { ...corsHeaders, "Content-Type": "application/json" },
      });
    }

    const body = await req.json().catch(() => ({}));
    emailId = body.email_id ?? null;
    if (!emailId) throw new Error("email_id is required");

    // CLAIM the row before sending. The conditional update is what makes a
    // retry safe: only a row still 'queued' flips to 'sending', so a duplicate
    // invocation (pg_net retry, double click, manual re-fire) finds nothing to
    // claim and sends nothing. A double-send is visible to the recipient and
    // there is no way to take it back.
    const { data: claimed, error: claimErr } = await supabase
      .from("admin_emails")
      .update({ status: "sending" })
      .eq("id", emailId)
      .eq("status", "queued")
      .select()
      .maybeSingle();

    if (claimErr) throw new Error(`claim failed: ${claimErr.message}`);
    if (!claimed) {
      return new Response(
        JSON.stringify({ success: true, skipped: "already claimed or not queued" }),
        { headers: { ...corsHeaders, "Content-Type": "application/json" } },
      );
    }

    const resendKey = Deno.env.get("RESEND_API_KEY");
    if (!resendKey) throw new Error("RESEND_API_KEY is not set in Edge Function secrets");

    const { data: nameCfg } = await supabase
      .from("system_configs").select("config_value")
      .eq("config_key", "ADMIN_EMAIL_FROM_NAME").maybeSingle();
    const fromName = nameCfg?.config_value || "KeepSIMs";

    const payload: Record<string, unknown> = {
      from: `${fromName} <${claimed.from_address}>`,
      to: claimed.to_addresses,
      subject: claimed.subject,
      html: renderHtml(claimed.body_text, fromName),
      // Always ship a text/plain alternative. Some corporate mail clients strip
      // HTML entirely, and an affiliate manager reading a blank email is worse
      // than a plain one.
      text: claimed.body_text,
    };
    if (claimed.cc_addresses?.length) payload.cc = claimed.cc_addresses;
    if (claimed.reply_to) payload.reply_to = claimed.reply_to;

    const res = await fetch("https://api.resend.com/emails", {
      method: "POST",
      headers: {
        "Content-Type": "application/json",
        Authorization: `Bearer ${resendKey}`,
      },
      body: JSON.stringify(payload),
    });

    const resBody = await res.json().catch(() => ({}));

    if (!res.ok) {
      // Resend's message is the useful part (unverified domain, bad address).
      // Surfacing it beats a generic failure the owner cannot act on.
      const msg = resBody?.message ?? resBody?.error ?? `HTTP ${res.status}`;
      await supabase.from("admin_emails")
        .update({ status: "failed", error: String(msg).slice(0, 500) })
        .eq("id", emailId);
      return new Response(JSON.stringify({ success: false, error: msg }), {
        status: 502,
        headers: { ...corsHeaders, "Content-Type": "application/json" },
      });
    }

    await supabase.from("admin_emails").update({
      status: "sent",
      provider_message_id: resBody?.id ?? null,
      sent_at: new Date().toISOString(),
      error: null,
    }).eq("id", emailId);

    return new Response(
      JSON.stringify({ success: true, id: resBody?.id ?? null }),
      { headers: { ...corsHeaders, "Content-Type": "application/json" } },
    );
  } catch (err) {
    const message = err instanceof Error ? err.message : String(err);
    // A row left in 'sending' would be stuck forever and unretryable, so any
    // failure after the claim must land it in a terminal state.
    if (emailId) {
      await supabase.from("admin_emails")
        .update({ status: "failed", error: message.slice(0, 500) })
        .eq("id", emailId).eq("status", "sending");
    }
    return new Response(JSON.stringify({ success: false, error: message }), {
      status: 500,
      headers: { ...corsHeaders, "Content-Type": "application/json" },
    });
  }
});
